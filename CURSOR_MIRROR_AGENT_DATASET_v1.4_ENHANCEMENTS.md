        raise FileNotFoundError(f"Schema not found: {schema_path}")
    
    with open(schema_path, 'r', encoding='utf-8') as f:
        return json.load(f)


def load_file(file_path: Path) -> Tuple[Dict, str]:
    """Load JSON or YAML file."""
    with open(file_path, 'r', encoding='utf-8') as f:
        if file_path.suffix == '.json':
            return json.load(f), 'json'
        elif file_path.suffix in ['.yaml', '.yml']:
            return yaml.safe_load(f), 'yaml'
        else:
            raise ValueError(f"Unsupported file type: {file_path.suffix}")


def validate_file(file_path: Path, schema: Dict) -> List[str]:
    """Validate file against schema. Returns list of errors."""
    errors = []
    
    try:
        data, file_type = load_file(file_path)
        validate(instance=data, schema=schema)
    except ValidationError as e:
        errors.append(f"Validation error in {file_path}: {e.message}")
        if e.path:
            errors.append(f"  Path: {'.'.join(str(p) for p in e.path)}")
    except SchemaError as e:
        errors.append(f"Schema error: {e.message}")
    except FileNotFoundError as e:
        errors.append(f"File not found: {e}")
    except Exception as e:
        errors.append(f"Unexpected error validating {file_path}: {e}")
    
    return errors
